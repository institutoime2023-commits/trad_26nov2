cmake_minimum_required(VERSION 3.22.1)
project(demucs_redmi_test LANGUAGES CXX)
set(SDENGINE_JNI ON CACHE BOOL "Build Demucs JNI" FORCE)
set(SDENGINE_TOOLS OFF CACHE BOOL "Disable desktop tools" FORCE)
set(SDENGINE_JNI_CLASS "com/orvex/demucstest/NativeEngine" CACHE STRING "JNI class" FORCE)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(DEMUCS_ENGINE_DIR "${PROJECT_ROOT}/third_party/demucs-android")
if(NOT EXISTS "${DEMUCS_ENGINE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Demucs engine source not found: ${DEMUCS_ENGINE_DIR}")
endif()
add_subdirectory("${DEMUCS_ENGINE_DIR}" demucs-engine)
